    def add_bound_constraint(self, var: Any, sign: SignEnum, value: int) -> list[Any]:
        """Add constraint of bound type on an integer variable (or expression) of the underlying cp model.

        `var` must compare to `value` according to `value`.

        Args:
            var:
            sign:
            value:

        Returns:

        """
        if sign == SignEnum.LEQ:
            cstr = var <= value
        elif sign == SignEnum.UEQ:
            cstr = var >= value
        elif sign == SignEnum.EQUAL:
            cstr = var == value
        elif sign == SignEnum.LESS:
            cstr = var < value
        elif sign == SignEnum.UP:
            cstr = var > value
        else:
            raise NotImplementedError(sign)
        return [self.add_bool_constraint(cstr)]

    def in_intervals(self, expr: Any, intervals: Iterable[tuple[int, int]]) -> Any:
        """Construct boolean expression true iff the given expr to be in given intervals."""
        return self.at_least_one(
            tuple(self.bool_and((lb <= expr, expr <= ub)) for lb, ub in intervals)

    @abstractmethod
    def add_allowed_assignments(
        self, variables: Sequence[Any], values: Iterable[Sequence[int]]
    ) -> list[Any]:
        """Enforce that variables evaluate to a sequence of values

        Args:
            variables:
            values: possible sequences for variables values

        Returns:
            created constraints

        """
        return [
            self.add_bool_constraint(
                self.at_least_one(
                    tuple(
                        self.bool_and(
                            tuple(var == value for var, value in zip(variables, row))
                        )
                        for row in values
                    )
                )
            )
        ]





[docs]
    @abstractmethod
    def add_forbidden_assignments(
        self,
        variables: Sequence[Any],
        values: Iterable[Sequence[int]],
    ) -> list[Any]:
        """Forbid sequence variables to evaluate in a sequence of values.

        A ForbiddenAssignments constraint is a constraint on an array of variables
        where the list of impossible combinations is provided in the tuples list.

        Args:
          variables: A list of variables.
          values: A list of forbidden tuples. Each tuple must have the same
            length as the variables, and the *i*th value of a tuple corresponds to
            the *i*th variable.

        Returns:
          the created constraints

        """
        return [
            self.add_bool_constraint(
                self.bool_and(
                    tuple(
                        self.at_least_one(
                            tuple(var != value for var, value in zip(variables, row))
                        )
                        for row in values
                    )
                )
            )
        ]

    def bool_and(self, exprs: Sequence[Any]) -> Any:
        """Construct conjunction of boolean expressions."""
        return self.sum(exprs) == len(exprs)

        """Adds a multiple circuit constraint, aka the 'VRP' constraint.

        The direct graph where arc #i (from tails[i] to head[i]) is present iff
        literals[i] is true must satisfy this set of properties:
        - #incoming arcs == 1 except for node 0.
        - #outgoing arcs == 1 except for node 0.
        - for node zero, #incoming arcs == #outgoing arcs.
        - There are no duplicate arcs.
        - Self-arcs are allowed except for node 0.
        - There is no cycle in this graph, except through node 0.

        Args:
          arcs: a list of arcs. An arc is a tuple (source_node, destination_node,
            literal). The arc is selected in the circuit if the literal is true.
            Both source_node and destination_node must be integers between 0 and the
            number of nodes - 1.

          the generated constraints
